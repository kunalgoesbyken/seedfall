//go:build wasip1

package z

import (
	"os"
	"syscall"
)

func mmap(fd *os.File, writeable bool, size int64) ([]byte, error) {
	if size == 0 {
		return nil, nil
	}
	prot := syscall.PROT_READ
	if writeable {
		prot |= syscall.PROT_WRITE
	}
	return syscall.Mmap(int(fd.Fd()), 0, int(size), prot, syscall.MAP_SHARED)
}

func munmap(b []byte) error {
	if len(b) == 0 {
		return nil
	}
	return syscall.Munmap(b)
}

func madvise(b []byte, readahead bool) error {
	return nil
}

func msync(b []byte) error {
	if len(b) == 0 {
		return nil
	}
	return syscall.Msync(b, syscall.MS_SYNC)
}
