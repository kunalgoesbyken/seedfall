const (
	seedfallAgain     = -1
	seedfallRefused   = -4
	seedfallInUse     = -5
	seedfallNoDest    = -7
	seedfallAddrMax   = 256
	seedfallConnFiles = "/.seedfall/conn/"
)

var (
	seedfallListeners sync.Map
	seedfallIPs       sync.Map
	seedfallSelfOnce  sync.Once
	seedfallSelfIP    IP
)

func seedfallSelf() IP {
	seedfallSelfOnce.Do(func() { seedfallSelfIP = seedfallLookupSelf() })
	return seedfallSelfIP
}

func socket(ctx context.Context, net string, family, sotype, proto int, ipv6only bool, laddr, raddr sockaddr, ctrlCtxFn func(context.Context, string, string, syscall.RawConn) error) (*netFD, error) {
	if sotype == syscall.SOCK_DGRAM && (family == syscall.AF_INET || family == syscall.AF_INET6) {
		return seedfallUDPFD(net, family, laddr, raddr)
	}
	if sotype != syscall.SOCK_STREAM {
		return fakeSocket(ctx, net, family, sotype, proto, ipv6only, laddr, raddr, ctrlCtxFn)
	}
	if raddr == nil {
		return seedfallListenFD(net, family, laddr)
	}
	return seedfallDialFD(net, family, raddr)
}

func seedfallLookupSelf() IP {
	name, _ := syscall.Getenv("SEEDFALL_NODE")
	return seedfallIP(name)
}

func seedfallIP(name string) IP {
	if ip, ok := seedfallIPs.Load(name); ok {
		return ip.(IP)
	}
	ip := IPv4(127, 0, 0, 1)
	if addrs, err := LookupHost(name); err == nil && len(addrs) > 0 {
		ip = ParseIP(addrs[0])
	}
	seedfallIPs.Store(name, ip)
	return ip
}

func seedfallSplit(addr string) (IP, int) {
	host, port, err := SplitHostPort(addr)
	if err != nil {
		return nil, 0
	}
	n, _ := strconv.Atoi(port)
	return seedfallIP(host), n
}

func seedfallEndpoint(fd *netFD, peer int32) (IP, int) {
	var buf [seedfallAddrMax]byte
	n := seedfallAddr(int32(fd.pfd.Sysfd), peer, unsafe.Pointer(&buf[0]), seedfallAddrMax)
	if n <= 0 {
		return nil, 0
	}
	return seedfallSplit(string(buf[:n]))
}

func seedfallCall(fn func(unsafe.Pointer, uint32) int32, s string) int32 {
	if s == "" {
		return fn(nil, 0)
	}
	b := []byte(s)
	return fn(unsafe.Pointer(&b[0]), uint32(len(b)))
}

func seedfallErr(op string, code int32) error {
	switch code {
	case seedfallRefused:
		return os.NewSyscallError(op, syscall.ECONNREFUSED)
	case seedfallInUse:
		return os.NewSyscallError(op, syscall.EADDRINUSE)
	case seedfallNoDest:
		return os.NewSyscallError(op, syscall.EDESTADDRREQ)
	}
	return os.NewSyscallError(op, syscall.EIO)
}

func seedfallTCPAddr(a sockaddr) *TCPAddr {
	if t, ok := a.(*TCPAddr); ok && t != nil {
		return t
	}
	return &TCPAddr{}
}

func seedfallHostPort(ip IP, port int) string {
	host := ""
	if len(ip) > 0 && !ip.IsUnspecified() {
		host = ip.String()
	}
	return JoinHostPort(host, strconv.Itoa(port))
}

func seedfallOpen(net string, family, sotype int, id int32) (*netFD, error) {
	sysfd, err := syscall.Open(seedfallConnFiles+strconv.Itoa(int(id)), syscall.O_RDWR, 0)
	if err != nil {
		return nil, os.NewSyscallError("open", err)
	}
	if err := syscall.SetNonblock(sysfd, true); err != nil {
		syscall.Close(sysfd)
		return nil, os.NewSyscallError("setnonblock", err)
	}
	seedfallBind(int32(sysfd), id)
	stream := sotype == syscall.SOCK_STREAM
	fd := newPollFD(net, poll.FD{Sysfd: sysfd, IsStream: stream, ZeroReadIsEOF: stream})
	fd.family, fd.sotype = family, sotype
	if err := fd.init(); err != nil {
		fd.Close()
		return nil, err
	}
	return fd, nil
}

func seedfallListenFD(net string, family int, laddr sockaddr) (*netFD, error) {
	local := seedfallTCPAddr(laddr)
	id := seedfallCall(seedfallListen, seedfallHostPort(local.IP, local.Port))
	if id < 0 {
		return nil, seedfallErr("listen", id)
	}
	fd, err := seedfallOpen(net, family, syscall.SOCK_STREAM, id)
	if err != nil {
		return nil, err
	}
	_, port := seedfallEndpoint(fd, 0)
	fd.setAddr(&TCPAddr{IP: seedfallSelf(), Port: port}, nil)
	seedfallListeners.Store(fd, id)
	return fd, nil
}

func seedfallDialFD(net string, family int, raddr sockaddr) (*netFD, error) {
	remote := seedfallTCPAddr(raddr)
	id := seedfallCall(seedfallDial, seedfallHostPort(remote.IP, remote.Port))
	if id < 0 {
		return nil, seedfallErr("connect", id)
	}
	fd, err := seedfallOpen(net, family, syscall.SOCK_STREAM, id)
	if err != nil {
		return nil, err
	}
	fd.isConnected = true
	_, port := seedfallEndpoint(fd, 0)
	fd.setAddr(&TCPAddr{IP: seedfallSelf(), Port: port}, remote)
	return fd, nil
}

func seedfallAccept(ln *netFD, lid int32) (*netFD, error) {
	var ready [1]byte
	for {
		if _, err := ln.pfd.Read(ready[:]); err != nil {
			return nil, err
		}
		id := seedfallAcceptID(lid)
		if id == seedfallAgain {
			continue
		}
		if id < 0 {
			return nil, seedfallErr("accept", id)
		}
		fd, err := seedfallOpen(ln.net, ln.family, syscall.SOCK_STREAM, id)
		if err != nil {
			return nil, err
		}
		fd.isConnected = true
		ip, port := seedfallEndpoint(fd, 1)
		fd.setAddr(ln.laddr, &TCPAddr{IP: ip, Port: port})
		return fd, nil
	}
}

func seedfallUDPAddr(a sockaddr) *UDPAddr {
	if u, ok := a.(*UDPAddr); ok && u != nil {
		return u
	}
	return &UDPAddr{}
}

func seedfallUDPFD(net string, family int, laddr, raddr sockaddr) (*netFD, error) {
	local := seedfallUDPAddr(laddr)
	id := seedfallCall(seedfallUDP, seedfallHostPort(local.IP, local.Port))
	if id < 0 {
		return nil, seedfallErr("bind", id)
	}
	fd, err := seedfallOpen(net, family, syscall.SOCK_DGRAM, id)
	if err != nil {
		return nil, err
	}
	_, port := seedfallEndpoint(fd, 0)
	self := &UDPAddr{IP: seedfallSelf(), Port: port}
	if raddr == nil {
		fd.setAddr(self, nil)
		return fd, nil
	}
	remote := seedfallUDPAddr(raddr)
	b := []byte(seedfallHostPort(remote.IP, remote.Port))
	if code := seedfallConnect(int32(fd.pfd.Sysfd), unsafe.Pointer(&b[0]), uint32(len(b))); code < 0 {
		fd.Close()
		return nil, seedfallErr("connect", code)
	}
	fd.isConnected = true
	fd.setAddr(self, remote)
	return fd, nil
}

func seedfallRecv(fd *netFD, p []byte) (int, *UDPAddr, error) {
	var from [seedfallAddrMax]byte
	var n int32
	var alen uint32
	err := fd.pfd.RawRead(func(s uintptr) bool {
		n = seedfallRecvFrom(int32(s), unsafe.Pointer(unsafe.SliceData(p)), uint32(len(p)), unsafe.Pointer(&from[0]), seedfallAddrMax, unsafe.Pointer(&alen))
		return n != seedfallAgain
	})
	if err != nil {
		return 0, nil, err
	}
	if n < 0 {
		return 0, nil, seedfallErr("recvfrom", n)
	}
	ip, port := seedfallSplit(string(from[:alen]))
	return int(n), &UDPAddr{IP: ip, Port: port}, nil
}

func seedfallSend(fd *netFD, sa syscall.Sockaddr, p []byte) (int, error) {
	addr := ""
	if sa != nil {
		if fd.isConnected {
			return 0, os.NewSyscallError("sendto", syscall.EISCONN)
		}
		to, _ := sockaddrToUDP(sa).(*UDPAddr)
		if to == nil {
			return 0, os.NewSyscallError("sendto", syscall.EAFNOSUPPORT)
		}
		addr = seedfallHostPort(to.IP, to.Port)
	}
	b := []byte(addr)
	var n int32
	err := fd.pfd.RawWrite(func(s uintptr) bool {
		n = seedfallSendTo(int32(s), unsafe.Pointer(unsafe.SliceData(b)), uint32(len(b)), unsafe.Pointer(unsafe.SliceData(p)), uint32(len(p)))
		return true
	})
	if err != nil {
		return 0, err
	}
	if n < 0 {
		return 0, seedfallErr("sendto", n)
	}
	return int(n), nil
}

func (fd *netFD) readFrom(p []byte) (int, syscall.Sockaddr, error) {
	if fd.fakeNetFD != nil {
		return fd.fakeNetFD.readFrom(p)
	}
	n, from, err := seedfallRecv(fd, p)
	if err != nil {
		return 0, nil, err
	}
	sa, err := from.sockaddr(fd.family)
	return n, sa, err
}

func (fd *netFD) readFromInet4(p []byte, sa *syscall.SockaddrInet4) (int, error) {
	if fd.fakeNetFD != nil {
		return fd.fakeNetFD.readFromInet4(p, sa)
	}
	n, from, err := seedfallRecv(fd, p)
	if err != nil {
		return 0, err
	}
	*sa, err = ipToSockaddrInet4(from.IP, from.Port)
	return n, err
}

func (fd *netFD) readFromInet6(p []byte, sa *syscall.SockaddrInet6) (int, error) {
	if fd.fakeNetFD != nil {
		return fd.fakeNetFD.readFromInet6(p, sa)
	}
	n, from, err := seedfallRecv(fd, p)
	if err != nil {
		return 0, err
	}
	*sa, err = ipToSockaddrInet6(from.IP, from.Port, "")
	return n, err
}

func (fd *netFD) readMsg(p, oob []byte, flags int) (int, int, int, syscall.Sockaddr, error) {
	if fd.fakeNetFD != nil {
		return fd.fakeNetFD.readMsg(p, oob, flags)
	}
	if flags != 0 {
		return 0, 0, 0, nil, os.NewSyscallError("readMsg", syscall.ENOTSUP)
	}
	n, sa, err := fd.readFrom(p)
	return n, 0, 0, sa, err
}
