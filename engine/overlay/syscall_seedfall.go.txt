//go:build wasip1

package syscall

import (
	"sync"
	"unsafe"
)

const (
	PROT_NONE  = 0x0
	PROT_READ  = 0x1
	PROT_WRITE = 0x2
	PROT_EXEC  = 0x4

	MAP_SHARED    = 0x1
	MAP_PRIVATE   = 0x2
	MAP_ANON      = 0x20
	MAP_ANONYMOUS = 0x20

	MS_ASYNC      = 0x1
	MS_INVALIDATE = 0x2
	MS_SYNC       = 0x4

	LOCK_SH = 0x1
	LOCK_EX = 0x2
	LOCK_NB = 0x4
	LOCK_UN = 0x8

	MADV_NORMAL     = 0x0
	MADV_RANDOM     = 0x1
	MADV_SEQUENTIAL = 0x2
	MADV_WILLNEED   = 0x3
	MADV_DONTNEED   = 0x4

	seedfallMapShared   = 0x1
	seedfallMapWritable = 0x2
	seedfallSyncSync    = 0x1
	seedfallLockRetry   = 1000000
)

type seedfallLock struct {
	exclusive bool
	holders   map[int]bool
}

var (
	seedfallMu    sync.Mutex
	seedfallMaps  = map[*byte][]byte{}
	seedfallLocks = map[uint64]*seedfallLock{}
)

//go:wasmimport seedfall mmap
func seedfallMmap(ino uint64, ptr unsafe.Pointer, size uint32, off uint64, flags uint32) uint32

//go:wasmimport seedfall msync
func seedfallMsync(ptr unsafe.Pointer, size uint32, flags uint32) uint32

//go:wasmimport seedfall munmap
func seedfallMunmap(ptr unsafe.Pointer) uint32

//go:linkname seedfallSleep time.Sleep
func seedfallSleep(ns int64)

func seedfallErr(code uint32) error {
	switch code {
	case 0:
		return nil
	case 1:
		return EBADF
	case 2:
		return EFAULT
	}
	return EIO
}

func Mmap(fd int, offset int64, length int, prot int, flags int) ([]byte, error) {
	if length <= 0 {
		return nil, EINVAL
	}
	b := make([]byte, length)
	if flags&MAP_ANON != 0 {
		return b, nil
	}
	var st Stat_t
	if err := Fstat(fd, &st); err != nil {
		return nil, err
	}
	var mode uint32
	if flags&MAP_SHARED != 0 {
		mode |= seedfallMapShared
		if prot&PROT_WRITE != 0 {
			mode |= seedfallMapWritable
		}
	}
	if err := seedfallErr(seedfallMmap(st.Ino, unsafe.Pointer(&b[0]), uint32(length), uint64(offset), mode)); err != nil {
		return nil, err
	}
	if flags&MAP_SHARED != 0 {
		seedfallMu.Lock()
		seedfallMaps[&b[0]] = b
		seedfallMu.Unlock()
	}
	return b, nil
}

func Munmap(b []byte) error {
	if len(b) == 0 {
		return EINVAL
	}
	seedfallMu.Lock()
	delete(seedfallMaps, &b[0])
	seedfallMu.Unlock()
	return seedfallErr(seedfallMunmap(unsafe.Pointer(&b[0])))
}

func Msync(b []byte, flags int) error {
	if len(b) == 0 {
		return nil
	}
	var mode uint32
	if flags&MS_SYNC != 0 {
		mode = seedfallSyncSync
	}
	return seedfallErr(seedfallMsync(unsafe.Pointer(&b[0]), uint32(len(b)), mode))
}

func Mlock(b []byte) error { return nil }

func Munlock(b []byte) error { return nil }

func Madvise(b []byte, advice int) error { return nil }

func Flock(fd int, how int) error {
	var st Stat_t
	if err := Fstat(fd, &st); err != nil {
		return err
	}
	if how&LOCK_UN != 0 {
		seedfallMu.Lock()
		defer seedfallMu.Unlock()
		if l := seedfallLocks[st.Ino]; l != nil {
			delete(l.holders, fd)
		}
		return nil
	}
	for !seedfallTryLock(st.Ino, fd, how&LOCK_EX != 0) {
		if how&LOCK_NB != 0 {
			return EAGAIN
		}
		seedfallSleep(seedfallLockRetry)
	}
	return nil
}

func seedfallTryLock(ino uint64, fd int, exclusive bool) bool {
	seedfallMu.Lock()
	defer seedfallMu.Unlock()
	l := seedfallLocks[ino]
	if l == nil {
		l = &seedfallLock{holders: map[int]bool{}}
		seedfallLocks[ino] = l
	}
	others := len(l.holders)
	if l.holders[fd] {
		others--
	}
	if others > 0 && (exclusive || l.exclusive) {
		return false
	}
	l.exclusive = exclusive
	l.holders[fd] = true
	return true
}
